```typescript
import type { Sensorium, StrategyProfile } from './sensorium.js'
import type { VigorState } from './vigor.js'

export interface TurnHookContext {
  cwd: string
  turn: number
  sensorium: Sensorium | null
  strategy: StrategyProfile | null
  vigor: VigorState
  recentToolHistory: Array<{ tool: string; status: string; target?: string }>
  gitChangeRate: number
  /** Mutable bag for hooks to communicate state changes back to the loop */
  mutations: {
    sensorium?: Sensorium
    strategy?: StrategyProfile
    vigor?: VigorState
    gitChangeRate?: number
    injectUserMessage?: string
    triggerThetaCheck?: boolean
  }
}

export interface ToolEvent {
  name: string
  success: boolean
  target?: string
}

export interface PreTurnHook {
  phase: 'preTurn'
  name: string
  run(ctx: TurnHookContext): Promise<void>
}

export interface PostToolHook {
  phase: 'postTool'
  name: string
  run(ctx: TurnHookContext, tool: ToolEvent): Promise<void>
}

export interface PostTurnHook {
  phase: 'postTurn'
  name: string
  run(ctx: TurnHookContext): Promise<void>
}

export type TurnHook = PreTurnHook | PostToolHook | PostTurnHook

export class HookPipeline {
  private preTurn: PreTurnHook[] = []
  private postTool: PostToolHook[] = []
  private postTurn: PostTurnHook[] = []

  constructor(hooks: TurnHook[]) {
    for (const h of hooks) {
      if (h.phase === 'preTurn') this.preTurn.push(h)
      else if (h.phase === 'postTool') this.postTool.push(h)
      else if (h.phase === 'postTurn') this.postTurn.push(h)
    }
  }

  async runPreTurn(ctx: TurnHookContext): Promise<void> {
    for (const h of this.preTurn) {
      try { await h.run(ctx) } catch { /* isolated */ }
    }
  }

  async runPostTool(ctx: TurnHookContext, tool: ToolEvent): Promise<void> {
    for (const h of this.postTool) {
      try { await h.run(ctx, tool) } catch { /* isolated */ }
    }
  }

  async runPostTurn(ctx: TurnHookContext): Promise<void> {
    for (const h of this.postTurn) {
      try { await h.run(ctx) } catch { /* isolated */ }
    }
  }
}
```

创建 `src/agent/hooks/sensorium-hook.ts`。此 hook 提取 loop.ts:514-607 的逻辑（git freshness 采样 + sensorium 计算 + provider health 混合 + strategy 计算 + theta interval 自适应 + star event + telemetry 写入）：

```typescript
import type { PreTurnHook, TurnHookContext } from '../turn-hooks.js'
import { getGitChangeRate, smoothChangeRate } from '../git-freshness.js'
import { computeSensorium, computeStrategy } from '../sensorium.js'
import type { SensoriumInput } from '../sensorium.js'
import { mapSensoriumToPhase, createStarEvent } from '../star-event.js'
import { join } from 'node:path'

export interface SensoriumHookDeps {
  getPressureResult: () => { ratio: number }
  getEvidenceState: () => { filesModified: number; verifiedCount: number }
  getToolHistory: () => string[]
  getPheromones: () => Array<{ path: string; signal: string; strength: number; depositedAt: number; halfLife: number }>
  getDoomLevel: () => 'none' | 'warn' | 'blocked'
  getPredictionAcc: () => { windowSize: number; predictions: boolean[]; consecutiveCorrect: number }
  getProviderDegradation: () => number
  getBaselineFingerprint: () => { combinedSha256: string } | null
  getCurrentFingerprint: () => { combinedSha256: string }
  onPhaseChange?: (phase: string, detail: { tool?: string; suggestion?: string }) => void
  hasEnteredHighComplexity: boolean
  isWriting: boolean
  isRunningTests: boolean
  isFinalTurn: boolean
  shouldEscalate: boolean
}

export function createSensoriumHook(deps: () => SensoriumHookDeps): PreTurnHook {
  return {
    phase: 'preTurn',
    name: 'sensorium',
    async run(ctx: TurnHookContext) {
      // Git freshness (fire-and-forget, updates ctx for next turn)
      getGitChangeRate(ctx.cwd).then(rate => {
        ctx.mutations.gitChangeRate = smoothChangeRate(rate, ctx.gitChangeRate)
      }).catch(() => {})

      const d = deps()
      const sensoriumInput: SensoriumInput = {
        predictionAcc: d.getPredictionAcc(),
        pressureResult: d.getPressureResult(),
        evidenceState: d.getEvidenceState(),
        toolCallHistory: d.getToolHistory(),
        pheromones: d.getPheromones() as any,
        doomLevel: d.getDoomLevel(),
        gitChangeRate: ctx.gitChangeRate,
      }
      let sensorium = computeSensorium(sensoriumInput)

      // Provider health → stability
      const degRatio = d.getProviderDegradation()
      if (degRatio > 0) {
        sensorium = { ...sensorium, stability: sensorium.stability * (1 - 0.3 * degRatio) }
      }

      const strategy = computeStrategy(sensorium)
      ctx.mutations.sensorium = sensorium
      ctx.mutations.strategy = strategy

      // Theta interval adaptation
      const baseInterval = strategy.thetaCycleInterval
      const gitMod = 1 - ctx.gitChangeRate * 0.5
      const adaptiveInterval = Math.max(2, Math.round(baseInterval * gitMod))

      // Star event
      const starCtx = {
        turn: ctx.turn,
        isWriting: d.isWriting,
        isRunningTests: d.isRunningTests,
        isFinalTurn: d.isFinalTurn,
        shouldEscalate: strategy.shouldEscalate,
        hasEnteredHighComplexity: d.hasEnteredHighComplexity,
      }
      const event = createStarEvent(sensorium, starCtx)
      if (d.onPhaseChange) {
        d.onPhaseChange(event.phase, { tool: event.glyph, suggestion: event.label })
      }

      // Telemetry
      const currentFP = d.getCurrentFingerprint()
      const baselineFP = d.getBaselineFingerprint()
      const driftEvent = baselineFP ? (currentFP.combinedSha256 !== baselineFP.combinedSha256) : false
      const telemetryLine = JSON.stringify({
        ts: Date.now(),
        turn: ctx.turn,
        phase: event.phase,
        ...sensorium,
        vigor: { tonic: ctx.vigor.tonic, phasic: ctx.vigor.phasic, curiosity: ctx.vigor.curiosity, vigor: ctx.vigor.vigor, variability: ctx.vigor.variability },
        strategy: { reasoningEffort: strategy.reasoningEffort, shouldEscalate: strategy.shouldEscalate, thetaInterval: adaptiveInterval },
        gitChangeRate: ctx.gitChangeRate,
        prefixDrift: driftEvent,
      })
      import('node:fs/promises').then(fs =>
        fs.appendFile(join(ctx.cwd, '.rivet', 'sensorium.jsonl'), telemetryLine + '\n', 'utf-8').catch(() => {})
      )
    },
  }
}
```

```typescript
import type { PreTurnHook, TurnHookContext } from '../turn-hooks.js'
import { shouldKick, buildKickActions, shouldEscalateFromKick } from '../dissipative-kick.js'
import type { StigmergyStore } from '../../context/stigmergy.js'

export function createKickHook(
  stigmergyStore: StigmergyStore,
  onPhaseChange?: (phase: string, detail: { reason?: string; suggestion?: string }) => void,
): PreTurnHook {
  return {
    phase: 'preTurn',
    name: 'dissipative-kick',
    async run(ctx: TurnHookContext) {
      const sensorium = ctx.mutations.sensorium ?? ctx.sensorium
      if (!sensorium) return
      if (!shouldKick(sensorium)) return

      const recentFailed = ctx.recentToolHistory
        .filter(h => h.status === 'failed')
        .map(h => h.target)
        .filter((t): t is string => Boolean(t))
      const kickActions = buildKickActions(sensorium, ctx.cwd, recentFailed)

      for (const path of kickActions.deadEndPaths) {
        stigmergyStore.deposit({ path, signal: 'dead-end', strength: 0.9 }).catch(() => {})
      }

      const fullMessage = kickActions.alternativeFrameworks.length > 0
        ? `${kickActions.injectedMessage}\n\n**替代框架：**\n${kickActions.alternativeFrameworks.map(f => `- ${f}`).join('\n')}`
        : kickActions.injectedMessage
      if (fullMessage) {
        ctx.mutations.injectUserMessage = fullMessage
      }
      if (shouldEscalateFromKick(sensorium) && onPhaseChange) {
        onPhaseChange('tianshu-encore', {
          reason: 'Dissipative kick: stagnation detected',
          suggestion: 'Escalate to stronger model or reframe the problem',
        })
      }
    },
  }
}
```

```typescript
import type { PostToolHook, TurnHookContext } from '../turn-hooks.js'
import { advanceThetaCounter, tickTheta, completeTheta } from '../star-event.js'
import type { ThetaState } from '../star-event.js'
import { runThetaCheck } from '../theta-check.js'
import type { RepairHintTracker } from '../repair-hint.js'

export function createThetaHook(
  getThetaState: () => ThetaState,
  setThetaState: (s: ThetaState) => void,
  repairHintTracker: RepairHintTracker,
): PostToolHook {
  return {
    phase: 'postTool',
    name: 'theta-gamma',
    async run(ctx: TurnHookContext) {
      let state = advanceThetaCounter(getThetaState())
      setThetaState(state)

      const sensorium = ctx.mutations.sensorium ?? ctx.sensorium
      if (!sensorium || sensorium.complexity <= 0.5) return
      if (!tickTheta(state, ctx.turn)) return

      runThetaCheck(ctx.cwd).then(result => {
        for (const errFile of result.errors) {
          repairHintTracker.recordFailure(errFile, 'type_error')
        }
      }).catch(() => {})
      setThetaState(completeTheta(state))
    },
  }
}
```

```typescript
import type { PostToolHook, TurnHookContext, ToolEvent } from '../turn-hooks.js'
import type { StigmergyStore } from '../../context/stigmergy.js'

export function createStigmergyHook(
  stigmergyStore: StigmergyStore,
  getEvidenceState: () => { verifications: Array<{ status: string }> },
  setLoadedPheromones: (p: any[]) => void,
): PostToolHook {
  return {
    phase: 'postTool',
    name: 'stigmergy',
    async run(ctx: TurnHookContext, tool: ToolEvent) {
      const evidence = getEvidenceState()

      if (tool.name === 'read_file' && tool.target) {
        const readCount = ctx.recentToolHistory.filter(
          h => h.tool === 'read_file' && h.target === tool.target
        ).length
        const hasWrite = ctx.recentToolHistory.some(
          h => (h.tool === 'write_file' || h.tool === 'edit_file') && h.target === tool.target
        )
        if (readCount >= 3 && !hasWrite) {
          stigmergyStore.deposit({ path: tool.target, signal: 'entry-point', strength: 0.4 }).catch(() => {})
        }
      }

      if ((tool.name === 'write_file' || tool.name === 'edit_file') && tool.target) {
        if (evidence.verifications.some(v => v.status === 'passed')) {
          stigmergyStore.deposit({ path: tool.target, signal: 'well-tested', strength: 0.6 }).catch(() => {})
        }
        if (evidence.verifications.some(v => v.status === 'failed')) {
          stigmergyStore.deposit({ path: tool.target, signal: 'fragile', strength: 0.8 }).catch(() => {})
        }
      }

      if (tool.name === 'bash') {
        const bashErrors = ctx.recentToolHistory.filter(h => h.tool === 'bash' && h.status === 'failed').length
        if (bashErrors >= 2) {
          const deadPath = tool.target ?? 'bash-command'
          stigmergyStore.deposit({ path: deadPath, signal: 'dead-end', strength: 0.9 }).catch(() => {})
        }
      }

      // Refresh pheromones after deposit
      stigmergyStore.query().then(p => {
        setLoadedPheromones(p.map(r => ({
          path: r.path, signal: r.signal, strength: r.currentStrength,
          depositedAt: r.depositedAt, halfLife: r.halfLife,
          ...(r.context ? { context: r.context } : {}),
        })))
      }).catch(() => {})
    },
  }
}
```

```typescript
import assert from 'node:assert/strict'
import { createVigorState, updateVigor, detectRigidity } from '../vigor.js'
import type { Sensorium } from '../sensorium.js'

function makeSensorium(overrides: Partial<Sensorium> = {}): Sensorium {
  return { momentum: 0.5, pressure: 0.3, confidence: 0.7, complexity: 0.3, freshness: 0.5, stability: 0.8, ...overrides }
}

describe('createVigorState', () => {
  it('returns neutral initial state', () => {
    const v = createVigorState()
```

```typescript
import type { Sensorium } from './sensorium.js'

export interface VigorState {
  tonic: number
  phasic: number
  curiosity: number
  vigor: number
  variability: number
  history: number[]
}

export function createVigorState(): VigorState {
  return { tonic: 0.5, phasic: 0, curiosity: 0, vigor: 0.5, variability: 0, history: [] }
}

export function updateVigor(prev: VigorState, toolSuccess: boolean, s: Sensorium): VigorState {
  const alpha = 0.15
  const actual = toolSuccess ? 1 : 0
  const tonic = alpha * actual + (1 - alpha) * prev.tonic
  const phasic = actual - prev.tonic
  const curiosity = (s.complexity > 0.5 && s.confidence < 0.4)
    ? Math.min(1, s.complexity - s.confidence)
    : 0
  const vigor = Math.max(0, Math.min(1, tonic + 0.3 * phasic + 0.2 * curiosity))
  const history = [...prev.history.slice(-9), vigor]
  const variability = history.length >= 5 ? computeStd(history) : 0
  return { tonic, phasic, curiosity, vigor, variability, history }
}

function computeStd(values: number[]): number {
  const mean = values.reduce((a, b) => a + b, 0) / values.length
  const variance = values.reduce((a, v) => a + (v - mean) ** 2, 0) / values.length
  return Math.sqrt(variance)
}

export function detectRigidity(history: number[], windowSize = 10, threshold = 0.05): boolean {
  if (history.length < windowSize) return false
  const recent = history.slice(-windowSize)
  return computeStd(recent) < threshold
}
```

```typescript
import type { PostToolHook, PreTurnHook, TurnHookContext, ToolEvent } from '../turn-hooks.js'
import { updateVigor, detectRigidity } from '../vigor.js'

export function createVigorPostToolHook(): PostToolHook {
  return {
    phase: 'postTool',
    name: 'vigor-update',
    async run(ctx: TurnHookContext, tool: ToolEvent) {
      const sensorium = ctx.mutations.sensorium ?? ctx.sensorium
      if (!sensorium) return
      const vigor = ctx.mutations.vigor ?? ctx.vigor
      ctx.mutations.vigor = updateVigor(vigor, tool.success, sensorium)
    },
  }
}

export function createVigorStrategyHook(): PreTurnHook {
  return {
    phase: 'preTurn',
    name: 'vigor-strategy',
    async run(ctx: TurnHookContext) {
      const vigor = ctx.mutations.vigor ?? ctx.vigor
      const strategy = ctx.mutations.strategy ?? ctx.strategy
      if (!strategy) return

      let adjusted = strategy

      // High vigor → lower effort (confident, move fast)
      if (vigor.vigor > 0.7 && strategy.reasoningEffort === 'medium') {
        adjusted = { ...adjusted, reasoningEffort: 'low' }
      }
      // Low vigor → higher effort (cautious, verify more)
      if (vigor.vigor < 0.3 && strategy.reasoningEffort === 'medium') {
        adjusted = { ...adjusted, reasoningEffort: 'high' }
      }
      // Negative RPE → raise commit threshold (alert mode)
      if (vigor.phasic < -0.5) {
        adjusted = { ...adjusted, commitThreshold: 0.95 }
      }

      // Rigidity detection → lower kick threshold (easier to break out)
      if (detectRigidity(vigor.history)) {
        const sensorium = ctx.mutations.sensorium ?? ctx.sensorium
        if (sensorium && sensorium.momentum < 0.3 && sensorium.stability < 0.4) {
          ctx.mutations.injectUserMessage = ctx.mutations.injectUserMessage ?? ''
        }
      }

      ctx.mutations.strategy = adjusted
    },
  }
}
```

```typescript
import type { PreTurnHook, TurnHookContext } from '../turn-hooks.js'
import type { StigmergyStore } from '../../context/stigmergy.js'
import { join } from 'node:path'

export function createEdgeProtectionHook(
  stigmergyStore: StigmergyStore,
  getSessionTurnCount: () => number,
  getUserAnchorText: () => string,
): PreTurnHook {
  let softLandingTriggered = false

  return {
    phase: 'preTurn',
    name: 'edge-protection',
    async run(ctx: TurnHookContext) {
      const sensorium = ctx.mutations.sensorium ?? ctx.sensorium
      if (!sensorium) return

      // Soft landing: pressure > 0.9 → write checkpoint + deposit handoff pheromone
      if (sensorium.pressure > 0.9 && !softLandingTriggered) {
        softLandingTriggered = true
        const checkpoint = {
          task: getUserAnchorText(),
          turn: getSessionTurnCount(),
          vigor: ctx.mutations.vigor ?? ctx.vigor,
          ts: Date.now(),
        }
        import('node:fs/promises').then(fs =>
          fs.writeFile(join(ctx.cwd, '.rivet', 'checkpoint.json'), JSON.stringify(checkpoint, null, 2), 'utf-8')
            .catch(() => {})
        )
        stigmergyStore.deposit({
          path: '__session__',
          signal: 'entry-point',
          strength: 0.9,
          context: `session-handoff turn ${getSessionTurnCount()}`,
        }).catch(() => {})
      }

      // ELM micro-release: vigor > 0.8 sustained → trigger early theta check
      const vigor = ctx.mutations.vigor ?? ctx.vigor
      const sustained = vigor.history.filter(v => v > 0.8).length >= 5
      if (vigor.vigor > 0.8 && sustained) {
        ctx.mutations.triggerThetaCheck = true
      }
    },
  }
}
```
