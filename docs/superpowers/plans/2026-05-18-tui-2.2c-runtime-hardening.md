## C1：Kick Cooldown

**问题**：stuck 状态持续时，dissipative kick 每个 turn 都注入 reframe message，导致 prompt 膨胀。

**文件**：`src/agent/hooks/kick-hook.ts`

**实现**：

```typescript
// 在 createKickRuntimeHook 中加 cooldown 状态
export interface KickRuntimeHookDeps {
  deposit: (deposit: PheromoneDeposit) => Promise<void>
  cooldownTurns?: number  // 默认 3
}

export function createKickRuntimeHook(deps: KickRuntimeHookDeps): PreTurnRuntimeHook {
  const cooldown = deps.cooldownTurns ?? 3
  let lastKickTurn = -Infinity

  return {
    phase: 'preTurn',
    name: 'dissipative-kick',
    run(ctx) {
      const { sensorium } = ctx.snapshot
      if (!sensorium || !shouldKick(sensorium)) return

      // Cooldown: 距离上次 kick 不足 N 个 turn，跳过
      const currentTurn = ctx.snapshot.turn
      if (currentTurn - lastKickTurn < cooldown) return

      lastKickTurn = currentTurn
      // ... 原有逻辑不变
    },
  }
}
```

**测试**：验证连续 stuck 时只在第 1 和第 4 个 turn 触发 kick（cooldown=3）。

**commit**：`fix(kick): add cooldown to prevent repeated reframe injection`

// src/agent/create-runtime-hooks.ts
import type { RuntimeHook } from './runtime-hooks.js'
import { createPerceptionRuntimeHook } from './hooks/perception-hook.js'
import { createKickRuntimeHook } from './hooks/kick-hook.js'
import { createVigorAfterPerceptionHook, createVigorPostToolHook } from './hooks/vigor-hook.js'
import { createThetaRuntimeHook } from './hooks/theta-hook.js'
import { createStigmergyRuntimeHook } from './hooks/stigmergy-hook.js'

export interface RuntimeHookDeps {
  stigmergyDeposit: (deposit: any) => Promise<void>
  stigmergyQuery: () => Promise<any>
  getEvidenceState: () => any
  setLoadedPheromones: (pheromones: any) => void
  getThetaState: () => any
  setThetaState: (state: any) => void
  getPredictionAccumulator: () => any

export function createDefaultRuntimeHooks(deps: RuntimeHookDeps): RuntimeHook[] {
  return [
    createPerceptionRuntimeHook(),
    createKickRuntimeHook({ deposit: deps.stigmergyDeposit }),
    createVigorAfterPerceptionHook(),
    createThetaRuntimeHook({
      getThetaState: deps.getThetaState,
      setThetaState: deps.setThetaState,

    createStigmergyRuntimeHook({
      deposit: deps.stigmergyDeposit,
      query: deps.stigmergyQuery,

      setLoadedPheromones: deps.setLoadedPheromones,
    }),
    createVigorPostToolHook({
      getPredictionAccumulator: deps.getPredictionAccumulator,
    }),
  ]
}

## C4：Telemetry Writer Seam

**问题**：sensorium.jsonl 写入逻辑内联在 loop.ts 第 668-694 行，包含 fingerprint drift 检测 + JSON 序列化 + 异步写文件。

**文件**：

**实现**：

```typescript
// src/agent/telemetry-writer.ts
import { join } from 'node:path'
import type { PerceptionTelemetrySnapshot } from './perception.js'

export interface TelemetryWriter {
  write(snapshot: PerceptionTelemetrySnapshot): void
}

export function createTelemetryWriter(cwd: string): TelemetryWriter {
  const path = join(cwd, '.rivet', 'sensorium.jsonl')
  return {
    write(snapshot) {
      const line = JSON.stringify(snapshot)
      import('node:fs/promises').then(fs =>
        fs.appendFile(path, line + '\n', 'utf-8').catch(() => {})
      )
    },
  }
}
```

**loop.ts 改动**：替换第 668-694 行为：

```typescript
this.telemetryWriter.write(buildTelemetrySnapshot({ ... }))
```

**可选**：注册为 postTurn hook（但 GPT-5.5 建议先做 seam，不急着变 hook）。

**测试**：无新测试（行为不变）。全量通过即可。

// /retrospect 命令处理
async function handleRetrospect(cwd: string): Promise<string> {
  const fs = await import('node:fs/promises')
  const sensoriumPath = join(cwd, '.rivet', 'sensorium.jsonl')
  const raw = await fs.readFile(sensoriumPath, 'utf-8').catch(() => '')
  if (!raw.trim()) return '无 sensorium 数据。请先运行一个 session。'

  const entries = parseSensoriumLog(raw)
  const report = generateRetrospect({
    sensoriumEntries: entries,
    gitLog: await getGitLog(cwd),
    toolEvents: [], // 从 trace-store 获取（如果可用）
    evidenceSummary: null,
    pheromones: null,
  })
  return report
}
