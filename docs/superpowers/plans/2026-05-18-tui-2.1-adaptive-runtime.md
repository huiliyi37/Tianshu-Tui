```typescript
import { spawn } from 'node:child_process'

export function getGitChangeRate(cwd: string, lookback = 5): Promise<number> {
  return new Promise(resolve => {
    const child = spawn('git', ['diff', '--stat', `HEAD~${lookback}`, '--numstat'], {
      cwd,
      stdio: ['ignore', 'pipe', 'pipe'],
      timeout: 100,
    })

    let stdout = ''
    child.stdout?.on('data', (chunk: Buffer) => { stdout += chunk.toString() })

    child.on('close', (code) => {
      if (code !== 0 || !stdout.trim()) { resolve(0); return }
      const changedFiles = stdout.trim().split('\n').filter(Boolean).length
      const child2 = spawn('git', ['ls-files'], { cwd, stdio: ['ignore', 'pipe', 'ignore'], timeout: 100 })
      let ls = ''
      child2.stdout?.on('data', (chunk: Buffer) => { ls += chunk.toString() })
      child2.on('close', () => {
        const total = ls.trim().split('\n').filter(Boolean).length
        resolve(total > 0 ? Math.min(1, changedFiles / total) : 0)
      })
      child2.on('error', () => resolve(0))
    })

    child.on('error', () => resolve(0))
  })
}

export function smoothChangeRate(raw: number, prev: number, alpha = 0.3): number {
  return Math.max(0, Math.min(1, alpha * raw + (1 - alpha) * prev))
}
```

```typescript
export type ProviderTier = 'hot' | 'warm' | 'cold'

interface ProviderState {
  tier: ProviderTier
  weight: number
  consecutiveFailures: number
  consecutiveSuccesses: number
}

export class ProviderHealthTracker {
  private providers: Map<string, ProviderState>

  constructor(providerIds: string[]) {
    this.providers = new Map(
      providerIds.map(id => [id, { tier: 'hot' as ProviderTier, weight: 0.5, consecutiveFailures: 0, consecutiveSuccesses: 0 }])
    )
  }

  recordSuccess(id: string): void {
    const p = this.providers.get(id)
    if (!p) return
    p.consecutiveSuccesses++
    p.consecutiveFailures = 0
    p.weight = Math.min(1, p.weight + 0.1 * (1 - p.weight))
    if (p.tier === 'warm' && p.consecutiveSuccesses >= 3) p.tier = 'hot'
    if (p.tier === 'cold' && p.consecutiveSuccesses >= 1) p.tier = 'warm'
  }

  recordFailure(id: string): void {
    const p = this.providers.get(id)
    if (!p) return
    p.consecutiveFailures++
    p.consecutiveSuccesses = 0
    p.weight = Math.max(0.05, p.weight - 0.4 * p.weight)
    if (p.tier === 'hot' && p.consecutiveFailures >= 2) p.tier = 'warm'
    if (p.tier === 'warm' && p.consecutiveFailures >= 3) p.tier = 'cold'
  }

  getTier(id: string): ProviderTier { return this.providers.get(id)?.tier ?? 'cold' }
  getWeight(id: string): number { return this.providers.get(id)?.weight ?? 0 }

  selectProvider(): string | undefined {
    const hot = [...this.providers.entries()].filter(([, s]) => s.tier === 'hot')
    if (hot.length > 0) return hot.sort((a, b) => b[1].weight - a[1].weight)[0][0]
    const warm = [...this.providers.entries()].filter(([, s]) => s.tier === 'warm')
    if (warm.length > 0) return warm.sort((a, b) => b[1].weight - a[1].weight)[0][0]
    const cold = [...this.providers.entries()].filter(([, s]) => s.tier === 'cold')
    return cold.length > 0 ? cold[0][0] : undefined
  }
}
```

```typescript
export interface SensoriumEntry {
  ts: number
  turn: number
  phase: string
  momentum: number
  pressure: number
  confidence: number
  complexity: number
  freshness: number
  stability: number
  strategy: { reasoningEffort: string; shouldEscalate: boolean; thetaInterval: number }
}

export function parseSensoriumLog(raw: string): SensoriumEntry[] {
  return raw.split('\n').filter(Boolean).map(line => {
    try { return JSON.parse(line) as SensoriumEntry }
    catch { return null }
  }).filter((e): e is SensoriumEntry => e !== null && typeof e.turn === 'number')
}

export function generateRetrospect(entries: SensoriumEntry[], gitLog: string[]): string {
  if (entries.length < 1) return '# Session Retrospective\n\n无足够数据，至少需要 1 个完整 turn。'

  const lines: string[] = ['# Session Retrospective\n']

  // Section 1: Factual Timeline
  lines.push('## 1. 事实时间线\n')
  lines.push(`| Turn | Phase | Momentum | Confidence | Stability | Strategy |`)
  lines.push(`|------|-------|----------|------------|-----------|----------|`)
  for (const e of entries) {
    lines.push(`| ${e.turn} | ${e.phase} | ${e.momentum.toFixed(2)} | ${e.confidence.toFixed(2)} | ${e.stability.toFixed(2)} | ${e.strategy.reasoningEffort} |`)
  }
  lines.push('')

  // Section 2: Four-layer analysis
  lines.push('## 2. 四层分析\n')
  const stabilityDrop = entries.some((e, i) => i > 0 && e.stability < entries[i - 1].stability - 0.2)
  const confidenceDrop = entries.some((e, i) => i > 0 && e.confidence < entries[i - 1].confidence - 0.3)
  const escalated = entries.some(e => e.strategy.shouldEscalate)

  lines.push('### L4 系统层')
  lines.push(escalated ? '- 触发了模型升级请求 — 检查任务是否超出当前模型能力边界' : '- 未触发升级，系统层无明显异常')
  lines.push('')

  lines.push('### L3 编排层')
  lines.push(entries.length > 10 ? '- 会话超过 10 turn — 检查任务拆解是否合理' : '- 会话长度正常')
  lines.push('')

  lines.push('### L2 上下文层')
  const maxPressure = Math.max(...entries.map(e => e.pressure))
  lines.push(maxPressure > 0.7 ? `- 上下文压力峰值 ${(maxPressure * 100).toFixed(0)}% — 可能影响输出质量` : '- 上下文压力正常')
  lines.push('')

  lines.push('### L1 执行层')
  lines.push(stabilityDrop ? '- 检测到 stability 显著下降 — 可能存在重复失败模式' : '- 执行稳定性正常')
  lines.push(confidenceDrop ? '- 检测到 confidence 显著下降 — 验证覆盖可能不足' : '')
  lines.push('')

  // Section 3: Root cause
  lines.push('## 3. 根因判定\n')
  if (stabilityDrop) {
    lines.push('- **Probable Cause**: stability 下降表明策略切换频繁或工具调用重复失败')
    lines.push('- **Contributing Factor**: 可能的上下文不足或任务复杂度超出预期')
  } else {
    lines.push('- 本次会话未检测到明显故障模式')
  }
  lines.push('')

  // Section 4: Recommendations
  lines.push('## 4. 寻址建议\n')
  if (stabilityDrop) lines.push('- **致系统设计**: 检查 doom loop 检测阈值是否过于敏感')
  if (confidenceDrop) lines.push('- **致用户**: 考虑在关键修改后手动运行测试验证')
  if (maxPressure > 0.7) lines.push('- **致工具维护**: 检查 compact 策略是否及时触发')
  if (!stabilityDrop && !confidenceDrop && maxPressure <= 0.7) lines.push('- 本次会话表现良好，无需特别调整')
  lines.push('')

  return lines.join('\n')
}
```
